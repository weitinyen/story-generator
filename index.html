// 真實 Gemini API 呼叫（自動匹配支援的模型版本） async function generateWithGemini(apiKey) { const e1 = document.getElementById('event_1').value; const f1 = document.getElementById('feeling_1').value; const e2 = document.getElementById('event_2').value; const f2 = document.getElementById('feeling_2').value; const e3 = document.getElementById('event_3').value; const f3 = document.getElementById('feeling_3').value; const prompt = `你是一位國中升學導師。請根據學生輸入的 1~3 步驟，推演 4~7 步驟故事。 1. 目標: ${e1} (感受: ${f1}) 2. 阻礙: ${e2} (感受: ${f2}) 3. 努力: ${e3} (感受: ${f3}) 要求： - 每格 (event, feeling) 限制為 1~3 句。 - 步驟 4~7 必須緊扣學生輸入的特定科目與做法，具備高潮迭起與因果連貫。 - scale 為 -3 到 +3 的數字字串。 請輸出標準 JSON 格式（切勿包含 Markdown 語語如 \`\`\`json）： { "steps": [ {"id":4, "event":"...", "feeling":"...", "scale":"2"}, {"id":5, "event":"...", "feeling":"...", "scale":"-3"}, {"id":6, "event":"...", "feeling":"...", "scale":"1"}, {"id":7, "event":"...", "feeling":"...", "scale":"3"} ] }`; const cleanedKey = apiKey.trim(); // 自動嘗試多個現有 API 模型，解決名稱變動導致的 404 問題 const candidateModels = ['gemini-2.5-flash', 'gemini-2.0-flash', 'gemini-1.5-flash-latest', 'gemini-2.5-pro']; let lastErrorMessage = ''; for (const modelName of candidateModels) { try { const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${modelName}:generateContent?key=${cleanedKey}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] }) }); const data = await response.json(); if (!response.ok) { lastErrorMessage = data.error?.message || `模型 ${modelName} 呼叫失敗`; continue; // 若此模型無效，自動切換下一個模型嘗試 } const rawText = data.candidates?.[0]?.content?.parts?.[0]?.text; if (!rawText) continue; const cleanJsonStr = rawText.replace(/```json/g, '').replace(/```/g, '').trim(); const jsonMatch = cleanJsonStr.match(/\{[\s\S]*\}/); if (jsonMatch) { const parsed = JSON.parse(jsonMatch[0]); parsed.steps.forEach(s => { if (document.getElementById(`event_${s.id}`)) { document.getElementById(`event_${s.id}`).value = s.event; document.getElementById(`feeling_${s.id}`).value = s.feeling; document.getElementById(`scale_${s.id}`).value = s.scale; } }); alert(`✨ 真實 Gemini AI (${modelName}) 動態推演成功！`); return; // 成功執行後即離開函式 } } catch (err) { lastErrorMessage = err.message; } } throw new Error(lastErrorMessage || "所有 Gemini 模型連線均未回應"); }
